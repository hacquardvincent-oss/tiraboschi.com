2 — JavaScript (à mettre dans tiraboschi.js)

/* ─── Newsletter Popup ─── */
(function() {
  const STORAGE_KEY   = 'tira_newsletter_v1';
  const AUTO_DELAY_MS = 8000;   // 8 secondes
  const SCROLL_RATIO  = 0.40;   // 40% de profondeur de scroll

  const popup    = document.getElementById('tira-popup');
  const backdrop = document.getElementById('tira-popup-backdrop');
  if (!popup) return;

  // ── Afficher
  function showPopup() {
    if (localStorage.getItem(STORAGE_KEY)) return;
    popup.classList.add('visible');
    backdrop.classList.add('visible');
    document.body.style.overflow = 'hidden';
  }

  // ── Fermer
  window.tiraNLClose = function(subscribed = false) {
    popup.classList.remove('visible');
    backdrop.classList.remove('visible');
    document.body.style.overflow = '';
    localStorage.setItem(STORAGE_KEY, subscribed ? 'subscribed' : 'closed');
  };

  // ── Trigger 1 : délai
  setTimeout(showPopup, AUTO_DELAY_MS);

  // ── Trigger 2 : scroll
  window.addEventListener('scroll', function onScroll() {
    if (window.scrollY / document.body.scrollHeight > SCROLL_RATIO) {
      showPopup();
      window.removeEventListener('scroll', onScroll);
    }
  }, { passive: true });

  // ── Soumission formulaire
  document.getElementById('popup-form')?.addEventListener('submit', async e => {
    e.preventDefault();
    const email = document.getElementById('popup-email').value;

    // POST vers Klaviyo via Shopify Customer API ou proxy
    // Remplacer par l'endpoint Klaviyo list subscribe
    await fetch('/contact#newsletter', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ contact: { email, tags: 'newsletter,atelier' } })
    });

    document.getElementById('popup-form-state').style.display = 'none';
    document.getElementById('popup-success-state').style.display = 'flex';
    localStorage.setItem(STORAGE_KEY, 'subscribed');
  });

  // ── Fermeture ESC
  document.addEventListener('keydown', e => {
    if (e.key === 'Escape') window.tiraNLClose();
  });
})();

3 — Logique localStorage (durée 30 jours)

// Version production avec expiry 30 jours
function setWithExpiry(key, value, days) {
  const expiry = new Date();
  expiry.setDate(expiry.getDate() + days);
  localStorage.setItem(key, JSON.stringify({ value, expiry: expiry.toISOString() }));
}

function getWithExpiry(key) {
  const item = localStorage.getItem(key);
  if (!item) return null;
  const parsed = JSON.parse(item);
  if (parsed.value === 'subscribed') return parsed.value; // jamais expirer
  if (new Date() > new Date(parsed.expiry)) {
    localStorage.removeItem(key);
    return null; // expiré → réafficher
  }
  return parsed.value;
}
