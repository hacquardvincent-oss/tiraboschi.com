01 — Composant

Smooth Scrolling (Lenis)

À ajouter dans theme.liquid — Phase 3B
Démo live

Scrollez cette page pour sentir le momentum. Lenis remplace le scroll natif du navigateur par une interpolation douce qui donne une sensation de profondeur et de matière — comme tourner les pages d'un livre plutôt que de les faire défiler mécaniquement.

La durée est réglée à 1.2s avec une courbe exponentielle qui décélère naturellement. Sur mobile, Lenis est désactivé pour respecter le scroll natif iOS/Android qui est déjà optimisé.

L'effet est subtil mais fondamental : c'est ce qui distingue un site de luxe d'un site e-commerce standard. Miu Miu, Jacquemus, The Row — tous utilisent un smooth scroll.

Le scroll stack de la homepage, les parallax et les reveal animations fonctionnent tous mieux avec Lenis car l'événement scroll est émis de façon fluide et continue.

theme.liquid<!-- Avant </body> dans theme.liquid -->
<script src="https://unpkg.com/@studio-freight/lenis@1.0.42/dist/lenis.min.js"></script>
<script>
// Initialisation Lenis smooth scroll
const lenis = new Lenis({
  duration:        1.2,
  easing:          (t) => Math.min(1, 1.001 - Math.pow(2, -10 * t)),
  orientation:     'vertical',
  smoothWheel:     true,
  touchMultiplier: 2
});

function raf(time) {
  lenis.raf(time);
  requestAnimationFrame(raf);
}
requestAnimationFrame(raf);

// Mobile : scroll natif iOS/Android est déjà optimal
if ('ontouchstart' in window) lenis.destroy();

// Exposer pour les autres scripts (ex. GSAP ScrollTrigger)
window.lenisInstance = lenis;
</script>

L'overlay blanc se pose sur la page, puis "disparaît" sur la page suivante — transition imperceptible mais luxueuse.

// CSS — dans tiraboschi.css
#tira-page-overlay {
  position: fixed; inset: 0; z-index: 9999;
  background: var(--tira-white);
  opacity: 0; pointer-events: none;
  transition: opacity 0.35s var(--tira-ease);
}
#tira-page-overlay.active { opacity: 1; pointer-events: all; }

// JS — dans tiraboschi.js
function initPageTransitions() {
  const overlay = document.getElementById('tira-page-overlay');
  if (!overlay) return;

  document.querySelectorAll('a[href^="/"]').forEach(link => {
    link.addEventListener('click', e => {
      // Ignorer : nouvelle tab, même page, ancres
      if (e.metaKey || e.ctrlKey) return;
      if (link.href === window.location.href) return;
      e.preventDefault();
      overlay.classList.add('active');
      setTimeout(() => window.location.href = link.href, 350);
    });
  });

  // Effacer l'overlay au retour (bfcache)
  window.addEventListener('pageshow', e => {
    if (e.persisted) overlay.classList.remove('active');
  });
}
document.addEventListener('DOMContentLoaded', initPageTransitions);

function initSplash() {
  if (sessionStorage.getItem('tira_splash_shown')) return;
  const splash = document.getElementById('tira-splash');
  splash.classList.add('active');
  document.body.style.overflow = 'hidden';
  setTimeout(() => {
    splash.classList.add('fade-out');
    document.body.style.overflow = '';
    sessionStorage.setItem('tira_splash_shown', '1');
  }, 2200);
}

Desktop uniquement — désactivé sur mobile pour ne pas perturber le touch 

tiraboschi.js// Ajouter class .tira-magnetic sur les CTAs principaux
// Fonctionne uniquement si hover est disponible (desktop)

function initMagneticButtons() {
  if (!window.matchMedia('(hover: hover)').matches) return;

  document.querySelectorAll('.tira-magnetic').forEach(btn => {
    btn.addEventListener('mousemove', e => {
      const rect = btn.getBoundingClientRect();
      const x = (e.clientX - rect.left - rect.width  / 2) * 0.3;
      const y = (e.clientY - rect.top  - rect.height / 2) * 0.3;
      btn.style.transform = `translate(${x}px, ${y}px)`;
      btn.style.transition = 'transform 0.15s ease';
    });

    btn.addEventListener('mouseleave', () => {
      btn.style.transform = '';
      btn.style.transition = 'transform 0.5s var(--tira-ease)';
    });
  });
}
document.addEventListener('DOMContentLoaded', initMagneticButtons);

sections/product-tiraboschi.liquid// Section "L'Atelier" — vidéo scrub sur la fiche produit
// La vidéo doit être encodée sans compression temporelle (mjpeg ou
// mp4 avec seekable metadata) pour permettre le seek instantané.

function initVideoScrub() {
  const video   = document.querySelector('.tira-scrub-video');
  const section = video?.closest('.tira-scrub-section');
  if (!video || !section) return;

  // Précharger les metadata pour connaître la durée
  video.preload = 'auto';

  let raf;
  function scrub() {
    const rect = section.getBoundingClientRect();
    const p = Math.max(0, Math.min(1,
      1 - rect.bottom / (rect.height + window.innerHeight)
    ));
    video.currentTime = video.duration * p;
  }

  const io = new IntersectionObserver(([entry]) => {
    if (entry.isIntersecting) {
      window.addEventListener('scroll', scrub, { passive: true });
    } else {
      window.removeEventListener('scroll', scrub);
    }
  }, { threshold: 0 });
  io.observe(section);
}

tiraboschi.js// Ajouter data-scheme="dark|grey|white" sur les sections de la homepage
// Le header réagit automatiquement

function initSchemeObserver() {
  const header = document.getElementById('shopify-section-header');
  if (!header) return;

  const sections = document.querySelectorAll('[data-scheme]');

  const io = new IntersectionObserver((entries) => {
    entries.forEach(entry => {
      if (entry.isIntersecting && entry.intersectionRatio > 0.5) {
        const scheme = entry.target.dataset.scheme;
        // Mettre à jour le header
        header.className = header.className
          .replace(/scheme-\w+/g, '')
          .trim();
        if (scheme !== 'white') {
          header.classList.add(`scheme-${scheme}`);
        }
        // Pour le logo : scheme-dark → logo blanc
        document.body.dataset.currentScheme = scheme;
      }
    });
  }, { threshold: 0.5 });

  sections.forEach(s => io.observe(s));
}
document.addEventListener('DOMContentLoaded', initSchemeObserver);
